### Null Safety

#### Null Safety Features

<KotlinPlayground 
  code={`fun main() {
    // Variables cannot be null by default
    var numberOfBooks: Int? = null
    println("Books: $numberOfBooks")
    
    // Safe call operator
    numberOfBooks = 6
    numberOfBooks = numberOfBooks?.dec()
    println("After decrement: $numberOfBooks")
    
    // Elvis operator
    numberOfBooks = null
    val finalCount = numberOfBooks?.dec() ?: 0
    println("Final count: $finalCount")
    
    // Safe call with string
    val name: String? = "Kotlin"
    println("Name length: \${name?.length}")
    
    val nullName: String? = null
    println("Null name length: \${nullName?.length}")
    
    // Not-null assertion (use carefully!)
    val definitelyNotNull: String? = "Hello"
    val length = definitelyNotNull!!.length
    println("Definite length: $length")
}`}
/>

### Error Handling with Result

Managing errors in Kotlin can be done using the `Result<T>` class, which allows you to handle success and failure cases without throwing exceptions.

<KotlinPlayground 
  code={`fun divide(a: Int, b: Int): Result<Double> {
    return if (b == 0) {
        Result.failure(Exception("Division par zéro"))
    } else {
        Result.success(a.toDouble() / b)
    }
}

fun main() {
    val result = divide(10, 2)
    
    result.fold(
        onSuccess = { value -> println("✅ Résultat: $value") },
        onFailure = { error -> println("❌ Erreur: \${error.message}") }
    )
    
    // Avec erreur
    divide(10, 0).fold(
        onSuccess = { value -> println("✅ Résultat: $value") },
        onFailure = { error -> println("❌ Erreur: \${error.message}") }
    )
}`}

#### Eager vs Lazy Filters

<KotlinPlayground 
  code={`fun main() {

}`}

## Advanced Data Modeling

### Data Classes vs Classes

Normal classes compare by reference, but in KMP we often need to compare data (like checking if the UI state changed).

<KotlinPlayground 
  code={`data class User(val name: String, val id: Int)
class UserClass(val name: String, val id: Int)

fun main() {
    val user1 = User("Alice", 1)
    val user2 = User("Alice", 1)
    
    val class1 = UserClass("Bob", 2)
    val class2 = UserClass("Bob", 2)
    
    // Data classes compare content automatically
    println("Data Class Equals: \${user1 == user2}") // true
    
    // Normal classes compare memory address
    println("Normal Class Equals: \${class1 == class2}") // false
    
    // Data classes have built-in copy
    val user3 = user1.copy(name = "Alice Smith")
    println(user3)
}`}
/>

### Sealed Classes for State

In Multiplatform development, strictly typing your UI state is crucial. Sealed classes act like "Super Enums" that can hold data.

<KotlinPlayground 
  code={`// Defines a restricted hierarchy
sealed class UiState {
    object Loading : UiState()
    data class Success(val data: String) : UiState()
    data class Error(val message: String) : UiState()
}

fun handleState(state: UiState) {
    // 'when' is exhaustive - compiler forces you to handle all cases
    when (state) {
        is UiState.Loading -> println("Spinner visible...")
        is UiState.Success -> println("Displaying: \${state.data}")
        is UiState.Error -> println("Ups! \${state.message}")
    }
}

fun main() {
    val currentState: UiState = UiState.Success("User Data Loaded")
    handleState(currentState)
    
    handleState(UiState.Error("No Internet"))
}`}
/>

## Concurrent Programming

### Overview

In this lesson, you'll learn about managing concurrent tasks in Kotlin:

- **Threads vs Coroutines**: Understanding the difference between heavy threads and lightweight coroutines
- **Suspend Functions**: The building blocks of coroutines
- **Coroutines Scopes**: managing the lifecycle of coroutines
- **Async/Await**: Handling asynchronous results
- **Dispatchers**: Controlling where code executes (Main, IO, Default)

### Coroutines Basics

#### Threads vs Coroutines

Coroutines are often called "lightweight threads". You can run thousands of coroutines on a few threads.

<KotlinPlayground 
  code={`import kotlinx.coroutines.*
import java.util.concurrent.atomic.AtomicInteger

fun main() = runBlocking {
    val c = AtomicInteger()
    
    // Launching 100,000 coroutines
    val jobs = List(100_000) {
        launch {
            c.incrementAndGet()
        }
    }
    
    jobs.forEach { it.join() }
    println("Completed \${c.get()} coroutines")
    
    // Trying this with Threads would likely crash with OutOfMemoryError
}`}
/>

#### Your First Coroutine

<KotlinPlayground 
  code={`import kotlinx.coroutines.*

fun main() = runBlocking { // this: CoroutineScope
    launch { // launch a new coroutine and continue
        delay(1000L) // non-blocking delay for 1 second
        println("World!")
    }
    println("Hello")
}`}
/>

### Suspend Functions

Suspend functions can pause their execution without blocking the thread, allowing other coroutines to run.

<KotlinPlayground 
  code={`import kotlinx.coroutines.*

// 'suspend' keyword is required to call other suspend functions like delay()
suspend fun doWorld() {
    delay(1000L)
    println("World!")
}

fun main() = runBlocking {
    launch {
        doWorld()
    }
    println("Hello")
}`}
/>

### Concurrency Patterns

#### Sequential vs Concurrent

<KotlinPlayground 
  code={`import kotlinx.coroutines.*
import kotlin.system.measureTimeMillis

suspend fun doSomethingUsefulOne(): Int {
    delay(1000L) // pretend we are doing something useful here
    return 13
}

suspend fun doSomethingUsefulTwo(): Int {
    delay(1000L) // pretend we are doing something useful here, too
    return 29
}

fun main() = runBlocking {
    // Sequential execution (takes ~2000ms)
    val time = measureTimeMillis {
        val one = doSomethingUsefulOne()
        val two = doSomethingUsefulTwo()
        println("The answer is \${one + two}")
    }
    println("Sequential completed in \$time ms")
    
    // Concurrent execution using async (takes ~1000ms)
    val timeAsync = measureTimeMillis {
        val one = async { doSomethingUsefulOne() }
        val two = async { doSomethingUsefulTwo() }
        println("The answer is \${one.await() + two.await()}")
    }
    println("Concurrent completed in \$timeAsync ms")
}`}
/>

#### Job Lifecycle

<KotlinPlayground 
  code={`import kotlinx.coroutines.*

fun main() = runBlocking {
    val job = launch {
        repeat(1000) { i ->
            println("job: I'm sleeping \$i ...")
            delay(500L)
        }
    }
    
    delay(1300L) // delay a bit
    println("main: I'm tired of waiting!")
    
    // Cancel the job
    job.cancel() 
    job.join() // waits for job's completion
    println("main: Now I can quit.")
}`}
/>
